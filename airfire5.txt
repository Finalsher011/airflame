local connections = {}
local function connect(signal, fn)
    local c = signal:Connect(fn)
    table.insert(connections, c)
    return c
end

local preferences = {}
local onChange = {}

local function setPref(key, value)
    preferences[key] = value
    if onChange[key] then onChange[key](value) end
end

local function pref(key, default)
    local v = preferences[key]
    if v == nil then return default end
    return v
end

local function prefNumber(key, default)
    local v = tonumber(pref(key, default))
    if v == nil or v ~= v or v == math.huge or v == -math.huge then return default end
    return v
end

local accentBindings = {}
local function bindAccent(inst, prop)
    table.insert(accentBindings, {inst, prop})
    inst[prop] = theme.accent
end

local function new(class, props, parent)
    local i = Instance.new(class)
    for k, v in pairs(props) do i[k] = v end
    if parent then i.Parent = parent end
    return i
end

local function corner(i, r)
    return new("UICorner", {CornerRadius = UDim.new(0, r)}, i)
end

local function tween(inst, props, t)
    if preferences["Interface Animation"] then
        TweenService:Create(inst, TweenInfo.new(t or 0.15, Enum.EasingStyle.Quad), props):Play()
    else
        for k, v in pairs(props) do inst[k] = v end
    end
end

local minimized = false
local function setMinimized(state)
    minimized = state
    if state then
        if preferences["Interface Animation"] then
            TweenService:Create(Main, TweenInfo.new(0.2, Enum.EasingStyle.Quad),
                {Size = UDim2.fromOffset(Config.WinW, Config.HeadH)}):Play()
            task.delay(0.21, function() if minimized then Body.Visible = false end end)
        else
            Main.Size = UDim2.fromOffset(Config.WinW, Config.HeadH)
            Body.Visible = false
        end
    else
        Body.Visible = true
        if preferences["Interface Animation"] then
            TweenService:Create(Main, TweenInfo.new(0.2, Enum.EasingStyle.Quad),
                {Size = UDim2.fromOffset(Config.WinW, Config.WinH)}):Play()
        else
            Main.Size = UDim2.fromOffset(Config.WinW, Config.WinH)
        end
    end
end

-- Drag (header) and slider drag share one InputChanged / InputEnded handler
local dragging, dragStart, startPos = false, nil, nil
local activeSlider = nil

Header.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = Main.Position
    end
end)

connect(UserInputService.InputChanged, function(input)
    local t = input.UserInputType
    if t ~= Enum.UserInputType.MouseMovement and t ~= Enum.UserInputType.Touch then return end
    if dragging then
        local d = input.Position - dragStart
        Main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X,
            startPos.Y.Scale, startPos.Y.Offset + d.Y)
    end
    if activeSlider then activeSlider(input) end
end)

connect(UserInputService.InputEnded, function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
        dragging = false
        activeSlider = nil
    end
end)

local function rowLabel(parent, text)
    return new("TextLabel", {
        Size = UDim2.new(0.5, 0, 1, 0), Position = UDim2.fromOffset(12, 0),
        BackgroundTransparency = 1, Text = text, TextColor3 = C.text,

local function buildToggle(item)
    local row = card(40)
    rowLabel(row, item.label)
    local pill = new("Frame", {
        Size = UDim2.fromOffset(40, 20), Position = UDim2.new(1, -52, 0.5, -10),
        BackgroundColor3 = C.off, BorderSizePixel = 0,
    }, row)
    corner(pill, 10)
    local knob = new("Frame", {
        Size = UDim2.fromOffset(16, 16), Position = UDim2.fromOffset(2, 2),
        BackgroundColor3 = Color3.new(1, 1, 1), BorderSizePixel = 0,
    }, pill)
    corner(knob, 8)

    local function render()
        local on = preferences[item.key] == true
        tween(pill, {BackgroundColor3 = on and theme.accent or C.off})
        tween(knob, {Position = on and UDim2.fromOffset(22, 2) or UDim2.fromOffset(2, 2)})
    end

    local hit = new("TextButton", {
        Size = UDim2.new(1, 0, 1, 0), BackgroundTransparency = 1, Text = "",
    }, row)
    hit.MouseButton1Click:Connect(function()
        setPref(item.key, not (preferences[item.key] == true))
        render()
    end)
    render()
end

local function buildSlider(item)
    local row = card(58)
    local label = new("TextLabel", {
        Size = UDim2.new(0.6, 0, 0, 26), Position = UDim2.fromOffset(12, 2),
        BackgroundTransparency = 1, Text = item.label, TextColor3 = C.text,
        Font = Enum.Font.Gotham, TextSize = 13, TextXAlignment = Enum.TextXAlignment.Left,
    }, row)
    local valueLabel = new("TextLabel", {
        Size = UDim2.new(0.3, 0, 0, 26), Position = UDim2.new(0.7, -12, 0, 2),
        BackgroundTransparency = 1, TextColor3 = theme.accent,
        Font = Enum.Font.GothamBold, TextSize = 13, TextXAlignment = Enum.TextXAlignment.Right,
    }, row)
    bindAccent(valueLabel, "TextColor3")

    local bar = new("Frame", {
        Size = UDim2.new(1, -24, 0, 6), Position = UDim2.fromOffset(12, 40),
        BackgroundColor3 = C.bg, BorderSizePixel = 0,
    }, row)
    corner(bar, 3)
    local fill = new("Frame", {Size = UDim2.new(0, 0, 1, 0), BorderSizePixel = 0}, bar)
    bindAccent(fill, "BackgroundColor3")
    corner(fill, 3)
    local knob = new("Frame", {
        Size = UDim2.fromOffset(14, 14), AnchorPoint = Vector2.new(0.5, 0.5),
        BackgroundColor3 = Color3.new(1, 1, 1), BorderSizePixel = 0,
    }, bar)
    corner(knob, 7)

    local function render()
        local v = prefNumber(item.key, item.value)
        local a = (v - item.min) / (item.max - item.min)
        fill.Size = UDim2.new(a, 0, 1, 0)
        knob.Position = UDim2.new(a, 0, 0.5, 0)
        valueLabel.Text = tostring(v)
    end

    local function fromInput(input)
        local w = bar.AbsoluteSize.X
        if w <= 0 then return end
        local a = math.clamp((input.Position.X - bar.AbsolutePosition.X) / w, 0, 1)
        local v = math.clamp(math.floor(item.min + a * (item.max - item.min) + 0.5), item.min, item.max)
        setPref(item.key, v)
        render()
    end

    local hit = new("TextButton", {
        Size = UDim2.new(1, -12, 0, 26), Position = UDim2.fromOffset(6, 30),
        BackgroundTransparency = 1, Text = "",
    }, row)
    hit.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            activeSlider = fromInput
            fromInput(input)
        end
    end)
    render()
end

local function buildChoice(item)
    local row = card(44)
    rowLabel(row, item.label)
    local opts = item.options

    if #opts <= 3 then
        local maxLen = 0
        for _, o in ipairs(opts) do maxLen = math.max(maxLen, #o) end
        local segW = math.max(60, maxLen * 7 + 18)
        local totalW = segW * #opts + 6
        local seg = new("Frame", {
            Size = UDim2.fromOffset(totalW, 28), Position = UDim2.new(1, -totalW - 10, 0.5, -14),
            BackgroundColor3 = C.bg, BorderSizePixel = 0,
        }, row)
        corner(seg, 8)
        new("UIListLayout", {FillDirection = Enum.FillDirection.Horizontal,
            SortOrder = Enum.SortOrder.LayoutOrder}, seg)
        new("UIPadding", {PaddingLeft = UDim.new(0, 3), PaddingTop = UDim.new(0, 3)}, seg)
        local buttons = {}
        local function render()
            for opt, b in pairs(buttons) do
                local sel = preferences[item.key] == opt
                tween(b, {BackgroundTransparency = sel and 0.1 or 1})
                b.TextColor3 = sel and Color3.new(1, 1, 1) or C.sub
            end
        end
        for i, opt in ipairs(opts) do
            local b = new("TextButton", {
                Size = UDim2.fromOffset(segW, 22), LayoutOrder = i, Text = opt,
                Font = Enum.Font.GothamMedium, TextSize = 12, AutoButtonColor = false,
                BackgroundTransparency = 1,
            }, seg)
            bindAccent(b, "BackgroundColor3")
            corner(b, 6)
            buttons[opt] = b
            b.MouseButton1Click:Connect(function()
                setPref(item.key, opt)
                render()
            end)
        end
        render()
    else
        local w = 200
        local box = new("Frame", {
            Size = UDim2.fromOffset(w, 28), Position = UDim2.new(1, -w - 10, 0.5, -14),
            BackgroundColor3 = C.bg, BorderSizePixel = 0,
        }, row)
        corner(box, 8)
        local valueLabel = new("TextLabel", {
            Size = UDim2.new(1, -56, 1, 0), Position = UDim2.fromOffset(28, 0),
            BackgroundTransparency = 1, Font = Enum.Font.GothamMedium, TextSize = 12,
        }, box)
        bindAccent(valueLabel, "TextColor3")
        local function index()
            for i, o in ipairs(opts) do if o == preferences[item.key] then return i end end
            return 1
        end
        local function render() valueLabel.Text = tostring(preferences[item.key]) end
        local function step(dir)
            local i = ((index() - 1 + dir) % #opts) + 1
            setPref(item.key, opts[i])
            render()
        end
        local function arrow(text, x, dir)
            local b = new("TextButton", {
                Size = UDim2.fromOffset(26, 28), Position = x, BackgroundTransparency = 1,
                Text = text, TextColor3 = C.sub, Font = Enum.Font.GothamBold, TextSize = 13,
            }, box)
            b.MouseButton1Click:Connect(function() step(dir) end)
        end
        arrow("<", UDim2.fromOffset(0, 0), -1)
        arrow(">", UDim2.new(1, -26, 0, 0), 1)
        render()
    end
end

-- Pure helpers

local function finiteVec(v)
    local lim = 1e6
    return v.X == v.X and v.Y == v.Y and v.Z == v.Z
        and math.abs(v.X) < lim and math.abs(v.Y) < lim and math.abs(v.Z) < lim
end

-- frame-rate independent; smoothness 1 = instant
local function smoothAlpha(smoothness, dt)
    local base = 1 / math.max(smoothness, 1)
    return 1 - (1 - base) ^ (math.clamp(dt, 0, 0.1) * 60)
end

local function leadPoint(pos, velocity, seconds)
    local p = pos + velocity * seconds
    if finiteVec(p) then return p end
    return pos
end

-- rotation (about the origin) that turns unit vector a onto unit vector b
local function rotationBetween(a, b)
    local axis = a:Cross(b)
    if axis.Magnitude < 1e-6 then return CFrame.new() end
    local angle = math.acos(math.clamp(a:Dot(b), -1, 1))
    return CFrame.fromAxisAngle(axis.Unit, angle)
end

local taggedModels = {}
local humanoidModels = {}
local scanned = false

    if inst:IsA("Model") then taggedModels[inst] = true end
end

    taggedModels[inst] = nil

local function addHumanoid(inst)
    if inst:IsA("Humanoid") then
        local m = inst.Parent
        if m and m:IsA("Model") then humanoidModels[m] = true end
    end
end

local function ensureScan()
    if scanned then return end
    scanned = true
    for _, inst in ipairs(Workspace:GetDescendants()) do addHumanoid(inst) end
    connect(Workspace.DescendantAdded, addHumanoid)
    connect(Workspace.DescendantRemoving, function(inst)
        if inst:IsA("Humanoid") and inst.Parent then humanoidModels[inst.Parent] = nil end
    end)
end

local function targetSet()
    if pref("Target Source", "Tagged") == "All Humanoids" then
        ensureScan()
        return humanoidModels
    end
    return taggedModels
end

local function readNpc(model)
    if not model.Parent then return nil end
    if Players:GetPlayerFromCharacter(model) then return nil end -- NPCs only
    if model == Player.Character then return nil end
    local hum = model:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then return nil end
    local root = model:FindFirstChild("HumanoidRootPart") or model.PrimaryPart
    if not root or not root:IsA("BasePart") then return nil end
    if not finiteVec(root.Position) then return nil end
    return {Model = model, Humanoid = hum, Root = root}
end

local function resolvePart(npc, names)
    for _, n in ipairs(names) do
        local p = npc.Model:FindFirstChild(n)
        if p and p:IsA("BasePart") then return p end
    end
    return npc.Root
end

local rayParams = RaycastParams.new()
rayParams.FilterType = Enum.RaycastFilterType.Exclude

local function isVisible(camPos, part, model)
    local ignore = {model}
    if Player.Character then table.insert(ignore, Player.Character) end
    rayParams.FilterDescendantsInstances = ignore
    return Workspace:Raycast(camPos, part.Position - camPos, rayParams) == nil
end

local function pickTarget(cam, fovPx, maxDist, names, needVisible)
    local camPos = cam.CFrame.Position
    local mouse = UserInputService:GetMouseLocation()
    local bestPart, bestModel, bestScore

    for model in pairs(targetSet()) do
        local npc = readNpc(model)
        if npc then
            local part = resolvePart(npc, names)
            local pos = part.Position
            if finiteVec(pos) and (pos - camPos).Magnitude <= maxDist then
                local sp = cam:WorldToViewportPoint(pos)
                if sp.Z > 0 then
                    local d = (Vector2.new(sp.X, sp.Y) - mouse).Magnitude
                    if d <= fovPx and (bestScore == nil or d < bestScore) then
                        if not needVisible or isVisible(camPos, part, model) then
                            bestPart, bestModel, bestScore = part, model, d
                        end
                    end
                end
            end
        end
    end

    if not bestPart then return nil end
    return {Part = bestPart, Model = bestModel, Position = bestPart.Position}
end

local function keyActive(id, bindName, modeName)
    local s = KeyState[id]
    if not s then s = {down = false, on = false}; KeyState[id] = s end
    local bind = BINDS[bindName]
    if not bind or not windowFocused or UserInputService:GetFocusedTextBox() then
        s.down = false; s.on = false
        return false
    end
    local down
    if bind.mouse then
        down = UserInputService:IsMouseButtonPressed(bind.mouse)
    else
        down = UserInputService:IsKeyDown(bind.key)
    end
    local result
    if modeName == "Toggle" then
        if down and not s.down then s.on = not s.on end
        result = s.on
    else
        s.on = false
        result = down
    end
    s.down = down
    return result
end

local function aimbotStep(dt)
    if not pref("Aimbot Enabled", false) then
        KeyState.aim = nil
        return
    end
    if not keyActive("aim", pref("Aim Key", "Right Mouse"), pref("Aim Mode", "Hold")) then return end

    local cam = Workspace.CurrentCamera
    if not cam then return end

    local names = {}
    for _, n in ipairs(AIM_PARTS[pref("Aim Part", "Head")] or AIM_PARTS.Head) do table.insert(names, n) end
    table.insert(names, "Head")

    local target = pickTarget(cam, prefNumber("FOV", 100), prefNumber("Target Distance", 1000),
        names, pref("Visible Check", false) == true)
    if not target then return end

    local lead = prefNumber("Prediction", 0) / 100 * Config.MaxLeadSeconds
    local aimPos = leadPoint(target.Position, target.Part.AssemblyLinearVelocity, lead)
    local camPos = cam.CFrame.Position
    if (aimPos - camPos).Magnitude < 1e-3 then return end

    cam.CFrame = cam.CFrame:Lerp(CFrame.lookAt(camPos, aimPos), smoothAlpha(prefNumber("Smoothness", 5), dt))
end

local function silentAim()
    if not pref("Silent Aim Enabled", false) then return nil end
    local cam = Workspace.CurrentCamera
    if not cam then return nil end

    if math.random() * 100 >= prefNumber("Hit Chance", 100) then return nil end
    local headshot = math.random() * 100 < prefNumber("Headshot Chance", 50)
    local names = headshot and {"Head"} or {"UpperTorso", "Torso"}

    local target = pickTarget(cam, prefNumber("Silent FOV", 150), prefNumber("Silent Distance", 1000),
        names, pref("Silent Visible Check", false) == true)
    if not target then return nil end
    return {Position = target.Position, Part = target.Part, Model = target.Model, Headshot = headshot}
end
shared.AFG_SilentAim = silentAim

-- Shot Redirect: rotate the camera so the ray through the cursor hits the target,
-- for this click only. Restored at the start of the next frame.
local pendingRestore = nil

connect(UserInputService.InputBegan, function(input, processed)
    if processed then return end
    if input.UserInputType ~= Enum.UserInputType.MouseButton1 then return end
    if not pref("Silent Aim Enabled", false) or pref("Silent Mode", "Shot Redirect") ~= "Shot Redirect" then return end
    if not windowFocused or UserInputService:GetFocusedTextBox() then return end

    local cam = Workspace.CurrentCamera
    if not cam then return end
    local hit = silentAim()
    if not hit then return end

    local camPos = cam.CFrame.Position
    local toTarget = hit.Position - camPos
    if toTarget.Magnitude < 1e-3 then return end

    local mouse = UserInputService:GetMouseLocation()
    local ray = cam:ViewportPointToRay(mouse.X, mouse.Y)
    local rot = rotationBetween(ray.Direction.Unit, toTarget.Unit)
    local originalCF = cam.CFrame
    local rotationOnly = originalCF - camPos
    pendingRestore = originalCF
    cam.CFrame = CFrame.new(camPos) * rot * rotationOnly
end)

local entries = {}

local function newLine()
    return new("Frame", {
        AnchorPoint = Vector2.new(0.5, 0.5), BorderSizePixel = 0, Visible = false,
    }, Overlay)
end

local function placeLine(f, a, b, thickness, color)
    local delta = b - a
    local len = delta.Magnitude
    if len ~= len or len < 0.5 or len > Config.MaxBoxPixels then f.Visible = false; return end
    f.Size = UDim2.fromOffset(len, thickness)
    f.Position = UDim2.fromOffset((a.X + b.X) / 2, (a.Y + b.Y) / 2)
    f.Rotation = math.deg(math.atan2(delta.Y, delta.X))
    f.BackgroundColor3 = color
    f.Visible = true
end

local function createEntry()
    local e = {Bones = {}}
    e.Box = new("Frame", {BackgroundTransparency = 1, BorderSizePixel = 0, Visible = false}, Overlay)
    e.Stroke = new("UIStroke", {}, e.Box)
    e.Label = new("TextLabel", {
        BackgroundTransparency = 1, Size = UDim2.fromOffset(240, 20), AnchorPoint = Vector2.new(0.5, 1),
        Font = Enum.Font.GothamMedium, TextStrokeTransparency = 0.4, Visible = false,
    }, Overlay)
    e.HealthBack = new("Frame", {
        BackgroundColor3 = C.bg, BorderSizePixel = 0, AnchorPoint = Vector2.new(1, 0), Visible = false,
    }, Overlay)
    e.HealthFill = new("Frame", {
        BorderSizePixel = 0, AnchorPoint = Vector2.new(0, 1), Position = UDim2.new(0, 0, 1, 0),
    }, e.HealthBack)
    e.Tracer = newLine()
    return e
end

local function hideEntry(e)
    e.Box.Visible = false
    e.Label.Visible = false
    e.HealthBack.Visible = false
    e.Tracer.Visible = false
    for _, b in ipairs(e.Bones) do b.Visible = false end
end

local function destroyEntry(e)
    e.Box:Destroy(); e.Label:Destroy(); e.HealthBack:Destroy(); e.Tracer:Destroy()
    for _, b in ipairs(e.Bones) do b:Destroy() end
end

local function projectBox(cam, cf, size)
    local minX, minY, maxX, maxY = math.huge, math.huge, -math.huge, -math.huge
    local any = false
    for ix = -1, 1, 2 do
        for iy = -1, 1, 2 do
            for iz = -1, 1, 2 do
                local corner3 = cf:PointToWorldSpace(Vector3.new(ix * size.X / 2, iy * size.Y / 2, iz * size.Z / 2))
                local sp = cam:WorldToViewportPoint(corner3)
                if sp.Z > 0 then
                    any = true
                    minX = math.min(minX, sp.X); maxX = math.max(maxX, sp.X)
                    minY = math.min(minY, sp.Y); maxY = math.max(maxY, sp.Y)
                end
            end
        end
    end
    if not any then return nil end
    if (maxX - minX) > Config.MaxBoxPixels or (maxY - minY) > Config.MaxBoxPixels then return nil end
    return Vector2.new(minX, minY), Vector2.new(maxX, maxY)
end

local function updateEntry(e, cam, npc, dist, o)
    local cf, size = npc.Model:GetBoundingBox()
    local tl, br = projectBox(cam, cf, size)
    if not tl then hideEntry(e); return end
    local w, h = br.X - tl.X, br.Y - tl.Y

    if o.box then
        e.Box.Position = UDim2.fromOffset(tl.X, tl.Y)
        e.Box.Size = UDim2.fromOffset(w, h)
        e.Stroke.Color = o.color
        e.Stroke.Thickness = o.thick
        e.Box.Visible = true
    else
        e.Box.Visible = false
    end

    local parts = {}
    if o.name then table.insert(parts, npc.Model.Name) end
    if o.distance then table.insert(parts, string.format("[%d st]", dist)) end
    if #parts > 0 then
        e.Label.Text = table.concat(parts, " ")
        e.Label.TextSize = o.textSize
        e.Label.TextColor3 = o.color
        e.Label.Position = UDim2.fromOffset((tl.X + br.X) / 2, tl.Y - 2)
        e.Label.Visible = true
    else
        e.Label.Visible = false
    end

    if o.health and npc.Humanoid.MaxHealth > 0 then
        local frac = math.clamp(npc.Humanoid.Health / npc.Humanoid.MaxHealth, 0, 1)
        e.HealthBack.Position = UDim2.fromOffset(tl.X - 4, tl.Y)
        e.HealthBack.Size = UDim2.fromOffset(o.thick + 1, h)
        e.HealthFill.Size = UDim2.new(1, 0, frac, 0)
        e.HealthFill.BackgroundColor3 = Color3.fromRGB(255, 0, 0):Lerp(Color3.fromRGB(0, 255, 0), frac)
        e.HealthBack.Visible = true
    else
        e.HealthBack.Visible = false
    end

    if o.tracer then
        placeLine(e.Tracer, Vector2.new(o.viewport.X / 2, o.viewport.Y),
            Vector2.new((tl.X + br.X) / 2, br.Y), o.thick, o.color)
    else
        e.Tracer.Visible = false
    end

    local bones = npc.Humanoid.RigType == Enum.HumanoidRigType.R15 and BONES_R15 or BONES_R6
    for i, pair in ipairs(bones) do
        local line = e.Bones[i]
        if not line then line = newLine(); e.Bones[i] = line end
        local drawn = false
        if o.skeleton then
            local a = npc.Model:FindFirstChild(pair[1])
            local b = npc.Model:FindFirstChild(pair[2])
            if a and b and a:IsA("BasePart") and b:IsA("BasePart") then
                local sa = cam:WorldToViewportPoint(a.Position)
                local sb = cam:WorldToViewportPoint(b.Position)
                if sa.Z > 0 and sb.Z > 0 then
                    placeLine(line, Vector2.new(sa.X, sa.Y), Vector2.new(sb.X, sb.Y), o.thick, o.color)
                    drawn = true
                end
            end
        end
        if not drawn then line.Visible = false end
    end
end

local function espStep()
    local cam = Workspace.CurrentCamera
    local o = {
        box = pref("Box ESP", false) == true,
        skeleton = pref("Skeleton ESP", false) == true,
        tracer = pref("Line ESP", false) == true,
        name = pref("Name ESP", false) == true,
        distance = pref("Distance", false) == true,
        health = pref("Health Bar", false) == true,
        thick = math.clamp(prefNumber("Esp Thickness", 2), 1, 5),
        textSize = math.clamp(math.floor(prefNumber("Text Size", 14)), 10, 24),
        color = ESP_COLORS[pref("ESP Color", "Cyan")] or ESP_COLORS.Cyan,
    }
    local anyOn = o.box or o.skeleton or o.tracer or o.name or o.distance or o.health
    local set = targetSet()

    local seen = {}
    if cam and anyOn then
        o.viewport = cam.ViewportSize
        local camPos = cam.CFrame.Position
        local maxDist = prefNumber("Max Distance", 1000)
        for model in pairs(set) do
            local npc = readNpc(model)
            if npc then
                local dist = (npc.Root.Position - camPos).Magnitude
                if dist <= maxDist and cam:WorldToViewportPoint(npc.Root.Position).Z > 0 then
                    seen[model] = true
                    local e = entries[model]
                    if not e then e = createEntry(); entries[model] = e end
                    updateEntry(e, cam, npc, dist, o)
                end
            end
        end
    end

    for model, e in pairs(entries) do
        if not set[model] or not model.Parent then
            destroyEntry(e)
            entries[model] = nil
        elseif not seen[model] then
            hideEntry(e)
        end
    end
end

local cleaned = false
local function cleanup()
    if cleaned then return end
    cleaned = true
    pcall(function() RunService:UnbindFromRenderStep(Config.MainStepName) end)
    pcall(function() RunService:UnbindFromRenderStep(Config.RestoreStepName) end)
    for _, c in ipairs(connections) do c:Disconnect() end
    for model, e in pairs(entries) do destroyEntry(e); entries[model] = nil end
    if shared.AFG_SilentAim == silentAim then shared.AFG_SilentAim = nil end
    if shared.AFG_Cleanup == cleanup then shared.AFG_Cleanup = nil end
    Overlay:Destroy()
    if GUI.Parent then GUI:Destroy() end
end
shared.AFG_Cleanup = cleanup
connect(GUI.Destroying, cleanup)

local reported = {}
local function guarded(tag, fn, ...)
    local ok, err = pcall(fn, ...)
    if not ok and not reported[tag] then
        reported[tag] = true
        warn("[AirFireGuns] " .. tag .. " error: " .. tostring(err))
    end
end

end)

end)
